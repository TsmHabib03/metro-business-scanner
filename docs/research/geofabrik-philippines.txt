   const layer = new GeoJSON(geojson, {
      style: { fillColor: '#ffcc66', color: '#ff9933', weight: 2, fillOpacity: 0.5 }
   }).addTo(map);
   map.fitBounds(layer.getBounds(), { padding: [20, 20] });
   const childrenGeo = {};
   
   

   let activeLayer = null;
   document.querySelectorAll('#subregions tr[data-child-id], #special_regions tr[data-child-id]').forEach(row => {
      row.addEventListener('mouseover', () => {
         const id = row.getAttribute('data-child-id');
         if (!childrenGeo[id]) return;
         if (activeLayer) map.removeLayer(activeLayer);
         activeLayer = new GeoJSON(childrenGeo[id], {
            style: { color: '#ff0000', weight: 0, fillOpacity: 0.6 }
         }).addTo(map);
      });
      row.addEventListener('mouseout', () => {
         if (activeLayer) { map.removeLayer(activeLayer); activeLayer = null; }
      });
   });
